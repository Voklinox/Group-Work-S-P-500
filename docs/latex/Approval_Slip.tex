\subsection{3. Managerial Research Question
(Problematic)}\label{managerial-research-question-problematic}

\begin{quote}
\textbf{``Does corporate governance risk (ISS QualityScore) compromise
financial profitability and market valuation among S\&P 500 firms, or
does strong governance provide a resilience and performance premium?''}
\end{quote}

\subsubsection{Alignment with the 4 Course
Sessions:}\label{alignment-with-the-4-course-sessions}

\begin{itemize}
\tightlist
\item
  \textbf{Session 1 (Univariate)}: What are the distribution profiles,
  central tendencies, and dispersions of profitability and governance
  risk? (Skewness, kurtosis, normality checks).
\item
  \textbf{Session 2 (Bivariate)}: Are high-risk governance firms
  disproportionately concentrated in certain sectors or size quartiles
  (Chi-Square)? Is governance risk correlated with profit margins
  (Pearson \(r\))?
\item
  \textbf{Session 3 (ANOVA)}: Does profitability or governance risk
  differ significantly across the 5 economic sectors (One-Way ANOVA \&
  Games-Howell/Tukey HSD)?
\item
  \textbf{Session 4 (Regression)}: Does governance risk predict profit
  margins when controlling for firm size (\texttt{Market\_Cap\_B}) and
  market volatility (\texttt{Beta})?
\end{itemize}
